public static List<Integer> extract_valid_ids(String[] rawData) {
    List<Integer> validIds = new ArrayList<>();
    int i = 0;
    while (i < rawData.length) {
        try {
            int id = Integer.parseInt(rawData[i]);
            if (id > 0) {
                validIds.add(id);
            }
        } catch (Exception e) {
            // skip the invalid ID
        }
        i++;
    }
    return validIds;
}

public static int fruit_distribution(String s, int n) {
    int result = n;
    for (String str : s.split(" ")) {
        int cnt = Integer.parseInt(str);
        result -= cnt;
    }
    return result;
}
